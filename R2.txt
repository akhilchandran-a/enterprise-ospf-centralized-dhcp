! ==========================================
! Project: Enterprise OSPF + Centralized DHCP
! Device: R2
! OSPF Process ID: 200
! ==========================================

! Enter configuration mode
enable
configure terminal

! BASIC CONFIGURATION
hostname R2

! Link to R1
interface GigabitEthernet0/0
 ip address 12.1.1.2 255.255.255.252
 no shutdown
 exit

! Link to R3
interface GigabitEthernet0/1
 ip address 23.1.1.1 255.255.255.252
 no shutdown
 exit

! Link to R4
interface GigabitEthernet0/2
 ip address 24.1.1.1 255.255.255.252
 no shutdown
 exit

! LAN10 Interface
interface GigabitEthernet0/3
 ip address 10.1.1.100 255.255.255.0
 ip helper-address 30.1.1.2
 no shutdown
 exit

! LOOPBACK
interface Loopback0
 ip address 2.2.2.2 255.255.255.255
 exit

! OSPF CONFIGURATION
router ospf 200
 router-id 2.2.2.2
 network 12.1.1.0 0.0.0.3 area 0
 network 23.1.1.0 0.0.0.3 area 0
 network 24.1.1.0 0.0.0.3 area 0
 network 2.2.2.2 0.0.0.0 area 0
 network 10.1.1.0 0.0.0.255 area 0
 exit

! DHCP RELAY CONFIGURATION 
interface GigabitEthernet0/3
 ip helper-address 30.1.1.2
 exit

end
write memory
! END OF R2 CONFIGURATION
