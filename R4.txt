! ==========================================
! Project: Enterprise OSPF + Centralized DHCP
! Device: R4
! OSPF Process ID: 400
! ==========================================

! Enter configuration mode
enable
configure terminal

! BASIC CONFIGURATION
hostname R4

! Link to R1
interface GigabitEthernet0/0
 ip address 14.1.1.2 255.255.255.252
 no shutdown
 exit

! Link to R2
interface GigabitEthernet0/1
 ip address 24.1.1.2 255.255.255.252
 no shutdown
 exit

! Link to R3
interface GigabitEthernet0/2
 ip address 34.1.1.2 255.255.255.252
 no shutdown
 exit

! LAN30 Interface
interface GigabitEthernet0/3
 ip address 30.1.1.100 255.255.255.0
 ip helper-address 30.1.1.2
 no shutdown
 exit

! LOOPBACK
interface Loopback0
 ip address 4.4.4.4 255.255.255.255
 exit

! OSPF CONFIGURATION
router ospf 400
 router-id 4.4.4.4
 network 14.1.1.0 0.0.0.3 area 0
 network 24.1.1.0 0.0.0.3 area 0
 network 34.1.1.0 0.0.0.3 area 0
 network 4.4.4.4 0.0.0.0 area 0
 network 30.1.1.0 0.0.0.255 area 0
 exit

! DHCP RELAY CONFIGURATION 
interface GigabitEthernet0/3
 ip helper-address 30.1.1.2
 exit


! END OF R4 CONFIGURATION
