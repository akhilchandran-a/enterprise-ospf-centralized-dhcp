! ==========================================
! Project: Enterprise OSPF + Centralized DHCP
! Device: DHCP-SERVER
! Role: Centralized DHCP Server
! ==========================================

! Enter configuration mode
enable
configure terminal

! BASIC CONFIGURATION
hostname SERVER

! INTERFACE CONFIGURATION
interface GigabitEthernet0/0
 ip address 30.1.1.2 255.255.255.0
 no shutdown
 exit

! DEFAULT ROUTE
! Forward traffic for remote networks to R4
ip route 0.0.0.0 0.0.0.0 30.1.1.100

! DHCP EXCLUDED ADDRESSES
! Reserve addresses for gateways and infrastructure
ip dhcp excluded-address 10.1.1.1 10.1.1.9
ip dhcp excluded-address 20.1.1.1 20.1.1.9
ip dhcp excluded-address 30.1.1.1 30.1.1.9

! DHCP POOL FOR LAN10
ip dhcp pool LAN10
 network 10.1.1.0 255.255.255.0
 default-router 10.1.1.100
 exit

! DHCP POOL FOR LAN20
ip dhcp pool LAN20
 network 20.1.1.0 255.255.255.0
 default-router 20.1.1.100
 exit

! DHCP POOL FOR LAN30
ip dhcp pool LAN30
 network 30.1.1.0 255.255.255.0
 default-router 30.1.1.100
 exit

! SAVE CONFIGURATION
end
write memory

! END OF DHCP SERVER CONFIGURATION
