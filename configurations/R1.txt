! ==========================================
! Project: Enterprise OSPF + Centralized DHCP
! Device: R1
! OSPF Process ID: 100
! ==========================================

! Enter configuration mode
enable
configure terminal

! BASIC CONFIGURATION
hostname R1

! Link to R2
interface GigabitEthernet0/0
ip address 12.1.1.1 255.255.255.252
no shutdown
exit

! Link to R4
interface GigabitEthernet0/1
ip address 14.1.1.1 255.255.255.252
no shutdown
exit

! Link to R3
interface GigabitEthernet0/2
ip address 13.1.1.1 255.255.255.252
no shutdown
exit

! LOOPBACK
interface Loopback0
ip address 1.1.1.1 255.255.255.255
exit

! OSPF CONFIGURATION
router ospf 100
router-id 1.1.1.1
network 12.1.1.0 0.0.0.3 area 0
network 13.1.1.0 0.0.0.3 area 0
network 14.1.1.0 0.0.0.3 area 0
network 1.1.1.1 0.0.0.0 area 0
exit

! END OF R1 CONFIGURATION
